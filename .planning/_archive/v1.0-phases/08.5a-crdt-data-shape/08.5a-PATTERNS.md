### `src/lib/collab/annotationSchema.ts` (REWRITE — converter/schema)

**Analog:** current file lines 59-102 (`codeMirrorToYjsAnnotation`) and 104-206 (`yjsAnnotationToCodeMirror`).

**New converter signature** (builds Y types directly, NOT JSON strings):
```typescript
/**
 * Convert a CodeMirror annotation to a Y.Map tree for sync.
 * CAUTION: each call MUST create fresh Y.Text / Y.Array / Y.Map instances —
 * a Y type can only be attached to one parent slot (Pitfall 2).
 */
export function codeMirrorToYjsAnnotation(
    annotation: GenericAnnotation,
    ytext: Y.Text,
    clientId: string,
): Y.Map<unknown> {
    const { startPos, endPos } = absoluteToRelative(ytext, annotation.selection);
    const ymap = new Y.Map<unknown>();
    ymap.set("id", generateAnnotationId(clientId));
    ymap.set("_type", annotation._type);
    ymap.set("startPos", startPos);
    ymap.set("endPos", endPos);

    const textNode = new Y.Text();
    ymap.set("text", textNode);
    const thread = new Y.Array<MessageObject>();
    for (const msg of annotation.thread) thread.push([toMessageObject(msg)]);
    ymap.set("thread", thread);
    ymap.set("annotations", new Y.Map());

    if (isAnnotationOfType(annotation, "suggestion")) {
        ymap.set("replacements", JSON.stringify(annotation.replacements));
        if (annotation.author) ymap.set("author", annotation.author);
    } else if (isAnnotationOfType(annotation, "revision")) {
        const versions = new Y.Map<Y.Map<unknown>>();
        for (const [idx, v] of annotation.versions.entries()) {
            const vMap = new Y.Map();
            const vText = new Y.Text();
            vText.insert(0, v.doc);
            vMap.set("text", vText);
            vMap.set("annotations", new Y.Map());
            if (v.label) vMap.set("label", v.label);
            versions.set(String(idx), vMap);
        }
        ymap.set("versions", versions);
        ymap.set("activeVersionIndex", annotation.activeVersionIndex);
    }
    return ymap;
}
```

### `src/lib/collab/types.ts` (REWRITE — type def)

**Analog:** current file lines 11-28.

**New type definition:**
```typescript
/** Yjs-stored annotation format for collaborative sync.
 *
 * Replaces the flat Phase 8 interface with a recursive Y.Map shape (D-90, D-92).
 * Each annotation is a Y.Map whose values are a mix of plain metadata and
 * Y shared types (text, thread, annotations, versions). TypeScript can't
 * fully express "a Y.Map with this key schema"; we document the shape in
 * comments and validate at boundaries.
 *
 * Y.Map layout:
 *   "id" -> string (client-prefixed)
 *   "_type" -> "comment" | "suggestion" | "revision"
 *   "startPos" -> Uint8Array (encoded RelativePosition)
 *   "endPos" -> Uint8Array
 *   "text" -> Y.Text (annotation-level text body, if editable)
 *   "thread" -> Y.Array<MessageObject>
 *   "annotations" -> Y.Map<YjsAnnotationNode>  (recursive)
 *   "replacements" -> string (JSON)  -- suggestion only
 *   "author" -> string               -- suggestion only
 *   "versions" -> Y.Map<string, Y.Map>  -- revision only (key = String(index))
 *   "activeVersionIndex" -> number      -- revision only
 *
 * Each version Y.Map has:
 *   "text" -> Y.Text
 *   "annotations" -> Y.Map<YjsAnnotationNode>
 *   "label" -> string | null
 */
export type YjsAnnotationNode = Y.Map<unknown>;

/** Thread message stored in Y.Array<MessageObject> (D-93).
 *
 * Shape MIRRORS CM `ThreadMessage` in `src/lib/editor/plugins/annotations/models.ts`
 * (`{ message, author, time }`) so the Y.Array payload is the CM thread payload with
 * no translation layer. Owned here (not re-exported from models.ts) to keep collab
 * types free of a circular edit→collab dependency.
 */
export interface MessageObject {
    message: string;
    author: string;
    time: number;
}

/** Active collab session state. `ymap` now holds YjsAnnotationNode values. */
export type CollabSession = {
    docId: string;
    clientID: string;
    isOwner: boolean;
    ydoc: Y.Doc;
    provider: WebsocketProvider;
    awareness: Awareness;
    ymap: Y.Map<YjsAnnotationNode>;
};
```

**Remove:** the old `YjsAnnotation` interface (flat shape with JSON strings).

### `src/lib/collab/yjsUndo.ts` (EXTEND — undo/extension)

**Analog:** current file lines 34-87 — keep the keymap structure; extend return shape.

**Pattern for adding `stack-item-popped` hookup and `addToScope` exposure:**
```typescript
import { keymap, type KeyBinding } from "@codemirror/view";
import * as Y from "yjs";
import type { Extension } from "@codemirror/state";

export function createYjsUndoExtension<T = unknown>(
    ytext: Y.Text,
    ymap?: Y.Map<T>,
): {
    extension: Extension;
    undoManager: Y.UndoManager;
} {
    const trackedTypes: (Y.Text | Y.Map<T>)[] = ymap ? [ytext, ymap] : [ytext];

    const undoManager = new Y.UndoManager(trackedTypes, {
        trackedOrigins: new Set(["local"]), // D-74
        captureTimeout: 500,                 // D-83, Pitfall 1+6
    });

    // D-97: isParentOf deep tracking means subtree Y.Texts under `ymap` are
    // already in scope. We still expose `addToScope` via the returned
    // undoManager so NestedEditorController can call it explicitly for clarity.

    const undoKeymap: KeyBinding[] = [
        { key: "Mod-z", run: () => { if (undoManager.canUndo()) { undoManager.undo(); return true; } return false; } },
        { key: "Mod-Shift-z", run: () => { if (undoManager.canRedo()) { undoManager.redo(); return true; } return false; } },
        { key: "Mod-y", run: () => { if (undoManager.canRedo()) { undoManager.redo(); return true; } return false; } },
    ];

    return { extension: keymap.of(undoKeymap), undoManager };
}
```

**Stack-item-popped listener** — lives in `collab/index.ts`, not here (keep yjsUndo.ts minimal).

### `src/lib/collab/test-helpers/twoPeerHarness.ts` (NEW — test harness)

**Analog:** duplicated `makePeer` + `connect` blocks at top of `yjsBinding.convergence.test.ts` (lines 15-55) and `yjsAnnotations.convergence.test.ts` (lines 22-79).

**Pattern to extract and promote:**
```typescript
/**
 * twoPeerHarness.ts — Shared 2-peer test fixture for collab convergence tests.
 *
 * Replaces duplicated makePeer/connect helpers across:
 *   - yjsBinding.convergence.test.ts
 *   - yjsAnnotations.convergence.test.ts
 *   - subtree-convergence.test.ts (new, Phase 8.5)
 *   - convergence-edgecases.test.ts (new, Phase 8.5)
 *   - recursive-mount.test.ts (new, Phase 8.5)
 */
import { EditorState } from "@codemirror/state";
import { EditorView } from "@codemirror/view";
import * as Y from "yjs";
import { createYjsBinding } from "../yjsBinding";
import { createAnnotationSyncPlugin } from "../yjsAnnotations";
import { annotationField } from "$lib/editor/plugins/annotations/annotationField";
import type { YjsAnnotationNode } from "../types";

export interface Peer {
    ydoc: Y.Doc;
    ytext: Y.Text;
    ymap: Y.Map<YjsAnnotationNode>;
    view: EditorView;
    clientId: string;
}

export function makePeer(clientId: string, initialText = ""): Peer {
    const ydoc = new Y.Doc();
    const ytext = ydoc.getText("document");
    const ymap = ydoc.getMap<YjsAnnotationNode>("annotations");
    if (initialText) {
        ydoc.transact(() => ytext.insert(0, initialText), "init");
    }
    const state = EditorState.create({
        doc: initialText,
        extensions: [
            annotationField,
            createYjsBinding(ytext),
            createAnnotationSyncPlugin(ytext, ymap, clientId),
        ],
    });
    const view = new EditorView({ state, parent: document.body });
    return { ydoc, ytext, ymap, view, clientId };
}

export function connect(a: Peer, b: Peer): () => void {
    const aToB = (update: Uint8Array, origin: unknown) => {
        if (origin === "remote") return;
        Y.applyUpdate(b.ydoc, update, "remote");
    };
    const bToA = (update: Uint8Array, origin: unknown) => {
        if (origin === "remote") return;
        Y.applyUpdate(a.ydoc, update, "remote");
    };
    a.ydoc.on("update", aToB);
    b.ydoc.on("update", bToA);
    Y.applyUpdate(b.ydoc, Y.encodeStateAsUpdate(a.ydoc), "remote");
    Y.applyUpdate(a.ydoc, Y.encodeStateAsUpdate(b.ydoc), "remote");
    return () => {
        a.ydoc.off("update", aToB);
        b.ydoc.off("update", bToA);
    };
}

export function teardown(peer: Peer): void {
    peer.view.destroy();
    peer.ydoc.destroy();
}
```
