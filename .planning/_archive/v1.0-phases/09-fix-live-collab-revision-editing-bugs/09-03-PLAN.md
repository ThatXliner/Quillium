```typescript
/**
 * Snapshot a version's Y.Text content into versions[versionIndex].doc (D-110).
 * Called before switching to a new version to ensure we have the latest content.
 */
private snapshotVersionYText(versionIndex: number): void {
    if (!this._hasCollabSubtree) return;
    
    const yjsId = getRevisionYjsId(this.revisionId);
    if (!yjsId) return;
    
    const ctx = getSubtreeContext(yjsId, versionIndex);
    if (!ctx) return;
    
    const currentYTextContent = ctx.subtreeYtext.toString();
    
    // Dispatch to update versions[versionIndex].doc with current Y.Text content
    this.parentView.dispatch({
        effects: [
            _updateRevisionVersionDoc.of({
                annotationId: this.revisionId,
                versionIndex,
                doc: currentYTextContent,
            }),
        ],
        annotations: [Transaction.addToHistory.of(false)],
    });
}
```
