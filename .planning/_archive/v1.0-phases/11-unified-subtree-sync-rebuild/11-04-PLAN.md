it("concurrent typing in same version merges character-by-character", async () => {
    // Owner creates a revision
    const revision = createNewRevision(1, 6, 11, "world");
    peerA.view.dispatch({
        effects: addAnnotation.of(revision),
    });
    
    await new Promise(resolve => queueMicrotask(resolve));
    
    // Get annotation IDs on both peers
    const annA = peerA.view.state.field(annotationField);
    const annIdA = Number(Object.keys(annA)[0]);
    const annB = peerB.view.state.field(annotationField);
    const annIdB = Number(Object.keys(annB)[0]);
    
    // Both peers type into the version concurrently
    // Peer A prepends "A"
    const trA = updateRevisionVersionState(peerA.view.state, annIdA, 0, "Aworld");
    peerA.view.dispatch(trA);
    
    // Peer B appends "B" (based on its current view which still has "world")
    // After A's change propagates, this should result in "AworldB"
    const trB = updateRevisionVersionState(peerB.view.state, annIdB, 0, "worldB");
    peerB.view.dispatch(trB);
    
    await new Promise(resolve => queueMicrotask(resolve));
    
    // Both should converge to the same merged result
    const finalA = peerA.view.state.field(annotationField);
    const finalB = peerB.view.state.field(annotationField);
    
    const revA = Object.values(finalA)[0];
    const revB = Object.values(finalB)[0];
    
    if (isAnnotationOfType(revA, "revision") && isAnnotationOfType(revB, "revision")) {
        // Both versions should have same text (merged)
        expect(revA.versions[0].doc).toBe(revB.versions[0].doc);
        // The merged text should contain both additions
        expect(revA.versions[0].doc).toContain("A");
        expect(revA.versions[0].doc).toContain("B");
        expect(revA.versions[0].doc).toContain("world");
    } else {
        expect.fail("Expected revision annotations");
    }
});
