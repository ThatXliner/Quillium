# Phase 7.5: Yjs Migration - Research

**Researched:** 2026-04-17
**Domain:** Real-time collaboration, CRDT synchronization, CodeMirror 6 integration
**Confidence:** HIGH

## Summary

This phase replaces the current OT-based collaboration stack (@codemirror/collab + Socket.io) with Yjs CRDT for more robust sync. Yjs provides automatic conflict resolution without version mismatch errors, native awareness protocol for cursors, and per-user undo via UndoManager with tracked origins. The migration is a clean replacement per D-70 -- no feature flags or gradual rollout.

The client requires a custom Yjs-CodeMirror 6 binding since y-codemirror.next is unmaintained (last publish: 2024-06-18 [VERIFIED: npm registry]). The relay server requires a complete rewrite using Yjs sync protocol (y-websocket compatible pattern) with custom authentication hooks. Persistence adapts from storing @codemirror/collab changesets to storing Yjs binary state updates via `Y.encodeStateAsUpdate()`.

**Primary recommendation:** Build a minimal custom binding (~200 LOC) using Y.Text directly with CodeMirror ViewPlugin pattern, implement y-websocket-compatible server with JWT auth in upgrade handler, and adapt existing Supabase persistence to store Yjs state vectors/updates as Uint8Array blobs.

### Pattern 1: Y.Text <-> CodeMirror Sync

**What:** Bidirectional sync between Y.Text shared type and CodeMirror EditorState
**When to use:** Every document edit (local or remote)
**Example:**

```typescript
// Source: [CITED: y-codemirror.next source, adapted]
import * as Y from 'yjs';
import { ViewPlugin, type ViewUpdate, type EditorView } from '@codemirror/view';
import { Annotation, Transaction } from '@codemirror/state';

// Annotation to mark transactions originating from Y.Text
const yjsAnnotation = Annotation.define<boolean>();

export function createYjsBinding(ytext: Y.Text) {
    return ViewPlugin.fromClass(class {
        private observer: (event: Y.YTextEvent, tr: Y.Transaction) => void;

        constructor(private view: EditorView) {
            // Y.Text -> CodeMirror
            this.observer = (event, yTransaction) => {
                if (yTransaction.local) return; // Skip local changes (already applied)
                
                const changes = [];
                let pos = 0;
                for (const delta of event.delta) {
                    if (delta.retain) {
                        pos += delta.retain;
                    } else if (delta.insert) {
                        const text = typeof delta.insert === 'string' ? delta.insert : '';
                        changes.push({ from: pos, to: pos, insert: text });
                        pos += text.length;
                    } else if (delta.delete) {
                        changes.push({ from: pos, to: pos + delta.delete, insert: '' });
                    }
                }
                
                if (changes.length > 0) {
                    this.view.dispatch({
                        changes,
                        annotations: [yjsAnnotation.of(true)],
                    });
                }
            };
            
            ytext.observe(this.observer);
        }

        update(update: ViewUpdate) {
            // CodeMirror -> Y.Text
            if (update.docChanged && !update.transactions.some(tr => tr.annotation(yjsAnnotation))) {
                const ydoc = ytext.doc;
                if (ydoc) {
                    ydoc.transact(() => {
                        update.changes.iterChanges((fromA, toA, fromB, toB, inserted) => {
                            if (toA > fromA) ytext.delete(fromA, toA - fromA);
                            if (inserted.length > 0) ytext.insert(fromA, inserted.toString());
                        });
                    }, 'local'); // Origin for UndoManager tracking
                }
            }
        }

        destroy() {
            ytext.unobserve(this.observer);
        }
    });
}
```

### Pattern 3: Per-User Undo via UndoManager

**What:** Track only local changes for undo, ignore remote changes
**When to use:** Preserve D-50 per-user undo behavior
**Example:**

```typescript
// Source: [CITED: docs.yjs.dev/api/undo-manager]
import * as Y from 'yjs';

const ydoc = new Y.Doc();
const ytext = ydoc.getText('document');
const undoManager = new Y.UndoManager(ytext, {
    trackedOrigins: new Set(['local']), // Only track local changes
    captureTimeout: 500, // Merge rapid changes into single undo step
});

// Apply local changes with 'local' origin
ydoc.transact(() => {
    ytext.insert(0, 'hello');
}, 'local');

// Remote changes have different origin (e.g., 'remote' or null) and won't be undone
undoManager.undo(); // Only undoes local 'hello' insert
```

### Pattern 4: Yjs Server Sync Handler

**What:** Handle WebSocket connections with Yjs sync protocol
**When to use:** Relay server connection handler
**Example:**

```typescript
// Source: [CITED: y-websocket/bin/utils.js pattern, adapted]
import * as Y from 'yjs';
import * as syncProtocol from 'y-protocols/sync';
import * as awarenessProtocol from 'y-protocols/awareness';
import * as encoding from 'lib0/encoding';
import * as decoding from 'lib0/decoding';

const messageSync = 0;
const messageAwareness = 1;

function handleConnection(ws: WebSocket, ydoc: Y.Doc, awareness: awarenessProtocol.Awareness) {
    // Send initial sync step 1
    const encoder = encoding.createEncoder();
    encoding.writeVarUint(encoder, messageSync);
    syncProtocol.writeSyncStep1(encoder, ydoc);
    ws.send(encoding.toUint8Array(encoder));

    ws.on('message', (data: Uint8Array) => {
        const decoder = decoding.createDecoder(data);
        const messageType = decoding.readVarUint(decoder);

        switch (messageType) {
            case messageSync:
                const encoder = encoding.createEncoder();
                encoding.writeVarUint(encoder, messageSync);
                const syncMessageType = syncProtocol.readSyncMessage(decoder, encoder, ydoc, null);
                if (encoding.length(encoder) > 1) {
                    ws.send(encoding.toUint8Array(encoder));
                }
                break;
            case messageAwareness:
                awarenessProtocol.applyAwarenessUpdate(awareness, decoding.readVarUint8Array(decoder), ws);
                break;
        }
    });
}
```

### Anti-Patterns to Avoid

- **Mixing OT and CRDT:** Never try to bridge @codemirror/collab state with Yjs. Clean migration only.
- **Storing Y.Doc as JSON:** Always use `Y.encodeStateAsUpdate()` for binary persistence. JSON loses CRDT metadata.
- **Global UndoManager:** Each client must have its own UndoManager with local tracked origin. Shared undo breaks per-user behavior.
- **Polling for updates:** Yjs uses push-based sync. Never implement pull loops like the current collabPlugin.
- **Ignoring awareness cleanup:** Must call `awareness.destroy()` and remove states on disconnect to prevent ghost cursors.

## Don't Hand-Roll

| Problem | Don't Build | Use Instead | Why |
|---------|-------------|-------------|-----|
| CRDT merge | Custom OT/CRDT algorithm | Yjs Y.Text | Battle-tested, handles all edge cases (concurrent insert at same position, etc.) |
| Sync protocol | Custom message format | y-protocols/sync | Standard protocol, handles state vectors, incremental updates, deduplication |
| Cursor positions | Absolute positions (current) | Y.RelativePosition | Survives document changes, no drift on concurrent edits |
| Binary encoding | JSON serialization | lib0/encoding | Efficient binary format, proper Uint8Array handling |
| Awareness lifecycle | Manual tracking | y-protocols/awareness | Handles timeout, cleanup, encoding automatically |

**Key insight:** Yjs ecosystem provides production-ready protocols. Custom implementations will miss edge cases around concurrent edits, network partitions, and position stability.

## Common Pitfalls

### Pitfall 1: Transaction Origin Confusion

**What goes wrong:** UndoManager undoes remote changes, or local changes aren't tracked
**Why it happens:** Forgetting to set origin on `doc.transact()`, or using wrong origin value
**How to avoid:** Always use `'local'` origin for user edits, and include it in `trackedOrigins` set. Remote changes should have `null` or `'remote'` origin.
**Warning signs:** Undo affects content user didn't type, or undo does nothing

### Pitfall 2: Yjs Update Duplication

**What goes wrong:** Document doubles content or shows corrupted state
**Why it happens:** Applying the same Yjs update multiple times
**How to avoid:** Yjs updates are idempotent but `Y.applyUpdate()` still needs care. Use state vectors to only send missing updates.
**Warning signs:** Document content duplicated, or garbage characters appear

### Pitfall 3: Awareness Memory Leaks

**What goes wrong:** Ghost cursors remain after users disconnect
**Why it happens:** Not cleaning up awareness states on WebSocket close
**How to avoid:** Call `awarenessProtocol.removeAwarenessStates(awareness, [clientID], 'disconnect')` in disconnect handler
**Warning signs:** Cursor decorations for users who left, increasing memory usage

### Pitfall 4: CodeMirror Transaction Loop

**What goes wrong:** Infinite loop of updates between CodeMirror and Y.Text
**Why it happens:** Y.Text observer dispatches to CodeMirror, CodeMirror update dispatches back to Y.Text
**How to avoid:** Use annotation to mark Y.Text-originated transactions, skip those in update handler
**Warning signs:** Browser freeze, stack overflow, extreme CPU usage

### Pitfall 5: Persistence Race Conditions

**What goes wrong:** Updates lost on server restart, or duplicate updates after restore
**Why it happens:** Persisting updates without proper sequencing, or not using state vectors for incremental loads
**How to avoid:** Use `Y.encodeStateVector()` to track what's persisted, `Y.encodeStateAsUpdate(doc, stateVector)` for incremental encoding
**Warning signs:** Document state jumps backwards on reconnect, missing recent edits

### Pitfall 6: Owner Disconnect with Yjs

**What goes wrong:** D-57 Live Room mode breaks -- collaborators keep editing after owner leaves
**Why it happens:** Yjs sync continues automatically, doesn't know about owner concept
**How to avoid:** Server tracks owner connection, broadcasts custom "ownerLeft" message, clients handle by disconnecting provider
**Warning signs:** Collaborators can edit after owner disconnected, no kick toast

### Persistence with State Vectors

```typescript
// Source: [CITED: docs.yjs.dev/api/document-updates, adapted]
import * as Y from 'yjs';

// Save document state to Postgres
async function persistYjsState(docId: string, ydoc: Y.Doc): Promise<void> {
    const update = Y.encodeStateAsUpdate(ydoc);
    const stateVector = Y.encodeStateVector(ydoc);
    
    // Store as Uint8Array (bytea in Postgres)
    await supabase.from('yjs_documents').upsert({
        document_id: docId,
        state_update: Buffer.from(update).toString('base64'), // Or use bytea
        state_vector: Buffer.from(stateVector).toString('base64'),
        updated_at: new Date().toISOString(),
    });
}

// Load document state from Postgres
async function loadYjsState(docId: string): Promise<Y.Doc> {
    const ydoc = new Y.Doc();
    
    const { data } = await supabase
        .from('yjs_documents')
        .select('state_update')
        .eq('document_id', docId)
        .single();
    
    if (data?.state_update) {
        const update = new Uint8Array(Buffer.from(data.state_update, 'base64'));
        Y.applyUpdate(ydoc, update);
    }
    
    return ydoc;
}
```

### WebSocket Server with Auth

```typescript
// Source: [CITED: y-websocket docs, adapted for Quillium]
import { createServer } from 'http';
import { WebSocketServer } from 'ws';
import { verifyJwt } from './auth/middleware';

const server = createServer();
const wss = new WebSocketServer({ noServer: true });

server.on('upgrade', async (request, socket, head) => {
    // Extract JWT from query params
    const url = new URL(request.url!, `http://${request.headers.host}`);
    const token = url.searchParams.get('auth');
    const docId = url.pathname.slice(1); // /docId -> docId
    
    try {
        const { userId, isOwner } = await verifyJwt(token, docId);
        
        wss.handleUpgrade(request, socket, head, (ws) => {
            // Attach user data to WebSocket
            (ws as any).userId = userId;
            (ws as any).isOwner = isOwner;
            wss.emit('connection', ws, request);
        });
    } catch (error) {
        socket.write('HTTP/1.1 401 Unauthorized\r\n\r\n');
        socket.destroy();
    }
});
```
