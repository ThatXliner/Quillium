```typescript
import { get } from "svelte/store";
import * as Y from "yjs";
import type { YjsAnnotationNode } from "./types";
// collabSession already imported

export function getRevisionYjsId(cmId: number): string | null {
    const session = get(collabSession);
    if (!session?.mainIdMap) return null;
    return session.mainIdMap.getYjsId(cmId) ?? null;
}

export function getSubtreeContext(
    revisionYjsId: string,
    versionIndex: number,
): {
    subtreeYtext: Y.Text;
    subtreeAnnotations: Y.Map<YjsAnnotationNode>;
    undoManager: Y.UndoManager;
} | null {
    const session = get(collabSession);
    if (!session) return null;
    const revNode = session.ymap.get(revisionYjsId);
    if (!(revNode instanceof Y.Map)) return null;
    const versionsMap = revNode.get("versions");
    if (!(versionsMap instanceof Y.Map)) return null;
    const versionNode = versionsMap.get(String(versionIndex));
    if (!(versionNode instanceof Y.Map)) return null;
    const subtreeYtext = versionNode.get("text");
    const subtreeAnnotations = versionNode.get("annotations");
    if (!(subtreeYtext instanceof Y.Text)) return null;
    if (!(subtreeAnnotations instanceof Y.Map)) return null;
    if (!session.undoManager) return null;
    return { subtreeYtext, subtreeAnnotations, undoManager: session.undoManager };
}
```

```typescript
/**
 * Signal whether the current nested EditorView must be torn down and rebuilt.
 *
 * Collab path (`_hasCollabSubtree === true`): observeDeep reconciles live, so
 *   the controller never needs a forced rebuild. The caller (Revision.svelte /
 *   RevisionModal.svelte) re-renders via the observer-driven CM dispatch path.
 *
 * Local-only path: compare the CURRENT version blob (JSON-serialised) against
 *   the blob we mounted. A mismatch means annotations or text shifted outside
 *   the live editor (e.g. a parent-level undo re-wrote the version); we must
 *   rebuild so the CM state matches the parent-authoritative blob again.

needsAnnotationRebuild(version: VersionState): boolean {
    if (this._hasCollabSubtree) return false; // observeDeep reconciles live
    const currentBlob = JSON.stringify(version);
    return currentBlob !== this._lastMountedBlob;
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| nested EditorView → parentView dispatch | Historically `translateAndDispatch` mapped nested doc deltas to parent coordinates. On the collab subtree path this cross-view pipe is bypassed; the subtree Y.Text is authoritative. Any regression where `translateAndDispatch` fires on the collab path would write stale JSON into the parent blob and clobber the CRDT state. |
| global UndoManager ↔ subtree Y.Text | Adding a subtree to the undo scope means the manager will capture local edits on it. If the subtree context resolves stale (e.g., after a version delete) we'd track a phantom type. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08.5-05-01 | Tampering | Stale `_hasCollabSubtree` flag after collab disconnects mid-session | mitigate | Recomputed at each `create()` call; sticky until destroy; reconnection forces a re-create via the existing `needsVersionSwitch` / modal lifecycle that already triggers destroy+create. |
| T-08.5-05-02 | Denial of Service | `hasSubtreeForRevision` called on every doc-change transaction | mitigate | Implementation is O(1) (Map lookup in collabSession + O(1) `Y.Map.get` walk). Even at 60 keystrokes/sec this is negligible compared to rendering cost. |
| T-08.5-05-03 | Repudiation | Edits made while collab connection was flapping silently fall into local-only flush path | accept | Standard Yjs behavior: offline edits queue into local ydoc updates and reconnect syncs them.
