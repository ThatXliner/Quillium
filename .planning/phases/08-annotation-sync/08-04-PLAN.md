```typescript
update(update: ViewUpdate) {
    // ... existing CM -> Y.Map sync code ...

    // After processing explicit effects, check for implicit revision changes
    // (version content changes via Phase 3 of annotationField)
    if (update.docChanged) {
        this.syncRevisionChanges(update);
    }
}

private syncRevisionChanges(update: ViewUpdate) {
    // Skip Yjs-originated transactions
    if (update.transactions.some((tr) => tr.annotation(yjsAnnotationSync))) {
        return;
    }

    const ydoc = ytext.doc;
    if (!ydoc) return;

    const oldAnnotations = update.startState.field(annotationField);
    const newAnnotations = update.state.field(annotationField);

    for (const [idStr, newAnn] of Object.entries(newAnnotations)) {
        if (!isAnnotationOfType(newAnn, "revision")) continue;

        const cmId = Number(idStr);
        const yjsId = this.idMap.getYjsId(cmId);
        if (!yjsId) continue;

        const existing = ymap.get(yjsId);
        if (!existing) continue;

        const oldAnn = oldAnnotations[cmId];
        if (!oldAnn || !isAnnotationOfType(oldAnn, "revision")) continue;

        // Check if activeVersionIndex changed
        const indexChanged = newAnn.activeVersionIndex !== oldAnn.activeVersionIndex;
        
        // Check if versions array changed (new version added, version deleted, content changed)
        const versionsChanged = JSON.stringify(newAnn.versions) !== JSON.stringify(oldAnn.versions);

        if (indexChanged || versionsChanged) {
            ydoc.transact(() => {
                ymap.set(yjsId!, {
                    ...existing,
                    versions: JSON.stringify(newAnn.versions),
                    activeVersionIndex: newAnn.activeVersionIndex,
                });
            }, "local");
        }
    }
}
```

```typescript
// In the observer callback, after handling add/update:
if (change.action === "update") {
    const yjsAnn = ymap.get(key);
    if (!yjsAnn) continue;

    const cmId = this.idMap.getCmId(key);
    if (cmId === undefined) continue;

    const currentAnnotations = this.view.state.field(annotationField);
    const existingAnn = currentAnnotations[cmId];
    if (!existingAnn) continue;

    // For revisions, check if activeVersionIndex or versions changed
    if (yjsAnn._type === "revision" && isAnnotationOfType(existingAnn, "revision")) {
        const remoteVersions: VersionState[] = yjsAnn.versions
            ? JSON.parse(yjsAnn.versions)
            : [];
        const remoteIndex = yjsAnn.activeVersionIndex ?? 0;

        // If activeVersionIndex changed and the version exists locally
        if (remoteIndex !== existingAnn.activeVersionIndex &&
            remoteVersions[remoteIndex] !== undefined) {
            // Dispatch version switch
            // Note: This uses addAnnotation to replace the annotation
            // A more surgical approach would be to dispatch the internal effects
            const updatedAnn: GenericAnnotation = {
                ...existingAnn,
                versions: remoteVersions,
                activeVersionIndex: remoteIndex,
            };
            effects.push(removeAnnotation.of(existingAnn));
            effects.push(addAnnotation.of(updatedAnn));
        }
    }
    
    // For all types, check if thread changed
    const remoteThread = JSON.parse(yjsAnn.thread);
    if (JSON.stringify(remoteThread) !== JSON.stringify(existingAnn.thread)) {
        // Thread update - but updateThread effect carries the new thread,
        // so we need to dispatch addAnnotation with updated thread
        const updatedAnn = { ...existingAnn, thread: remoteThread };
        effects.push(removeAnnotation.of(existingAnn));
        effects.push(addAnnotation.of(updatedAnn));
    }
}
```

```typescript
describe("revision sync", () => {
    function createRevisionAnnotation(
        id: number,
        from: number,
        to: number,
        versions: { doc: string }[],
        activeVersionIndex: number = 0,
    ): GenericAnnotation {
        return {
            id,
            _type: "revision",
            selection: EditorSelection.single(from, to),
            thread: [],
            versions,
            activeVersionIndex,
        };
    }

    it("syncs activeVersionIndex change to Y.Map", () => {
        const annotation = createRevisionAnnotation(0, 0, 5, [
            { doc: "hello" },
            { doc: "world" },
        ], 0);

        view.dispatch({
            effects: [addAnnotation.of(annotation)],
        });

        // Get Yjs ID
        const yjsId = Array.from(ymap.keys())[0];
        expect(ymap.get(yjsId)!.activeVersionIndex).toBe(0);

        // Update to switch version (simulating what happens after setActiveRevisionVersion)
        const updatedAnnotation = { ...annotation, activeVersionIndex: 1 };
        view.dispatch({
            effects: [
                removeAnnotation.of(annotation),
                addAnnotation.of(updatedAnnotation),
            ],
        });

        const yjsAnn = ymap.get(yjsId)!;
        expect(yjsAnn.activeVersionIndex).toBe(1);
    });

    it("syncs new version addition to Y.Map", () => {
        const annotation = createRevisionAnnotation(0, 0, 5, [{ doc: "hello" }], 0);

        view.dispatch({
            effects: [addAnnotation.of(annotation)],
        });

        const yjsId = Array.from(ymap.keys())[0];
        const initialVersions = JSON.parse(ymap.get(yjsId)!.versions!);
        expect(initialVersions).toHaveLength(1);

        // Add a new version
        const updatedAnnotation = {
            ...annotation,
            versions: [{ doc: "hello" }, { doc: "new version" }],
            activeVersionIndex: 1,
        };
        view.dispatch({
            effects: [
                removeAnnotation.of(annotation),
                addAnnotation.of(updatedAnnotation),
            ],
        });

        const yjsAnn = ymap.get(yjsId)!;
        const versions = JSON.parse(yjsAnn.versions!);
        expect(versions).toHaveLength(2);
        expect(versions[1].doc).toBe("new version");
    });

    it("receives remote activeVersionIndex change", () => {
        // Add annotation locally
        const annotation = createRevisionAnnotation(0, 0, 5, [
            { doc: "hello" },
            { doc: "world" },
        ], 0);

        view.dispatch({
            effects: [addAnnotation.of(annotation)],
        });

        const yjsId = Array.from(ymap.keys())[0];
        const existing = ymap.get(yjsId)!;

        // Simulate remote version switch
        ydoc.transact(() => {
            ymap.set(yjsId, {
                ...existing,
                activeVersionIndex: 1,
            });
        }, "remote");

        // Check CM state updated
        const annotations = view.state.field(annotationField);
        const updated = annotations[0];
        if (isAnnotationOfType(updated, "revision")) {
            expect(updated.activeVersionIndex).toBe(1);
        }
    });
});

describe("thread sync", () => {
    it("receives remote thread reply", () => {
        // Add comment locally
        const annotation: GenericAnnotation = {
            id: 0,
            _type: "comment",
            selection: EditorSelection.single(0, 5),
            thread: [{ message: "initial", author: "user1", time: 100 }],
        };

        view.dispatch({
            effects: [addAnnotation.of(annotation)],

        const yjsId = Array.from(ymap.keys())[0];

        // Simulate remote reply
        const updatedThread = [
            { message: "initial", author: "user1", time: 100 },
            { message: "reply", author: "user2", time: 200 },

        ydoc.transact(() => {
            ymap.set(yjsId, {
                ...existing,
                thread: JSON.stringify(updatedThread),

        // Check CM state has the reply

        const updated = annotations[0];
        expect(updated.thread).toHaveLength(2);

        expect(thread).toHaveLength(1);

    });
});
```
